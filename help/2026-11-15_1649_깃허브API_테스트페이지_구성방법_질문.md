## 2. `index.html` — 최소 뼈대

```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>GitHub API 테스트</title>
    <link rel="stylesheet" href="style.css">
    <script src="script.js" defer></script>
</head>
<body>
    <main>
        <h1>GitHub 저장소 테스트</h1>

        <!-- 레이트리밋(시간당 60회)을 아끼기 위해 버튼 눌렀을 때만 요청 -->
        <div class="controls">
            <input type="text" id="username-input" placeholder="깃허브 아이디 입력" value="본인아이디">
            <button type="button" id="load-btn">불러오기</button>
        </div>

        <!-- 4가지 상태(로딩 / 에러 / 빈 상태) 표시 영역 -->
        <div id="status" class="status"></div>

        <!-- 성공 시 카드 목록이 들어갈 영역 -->
        <div id="project-grid" class="project-grid"></div>
    </main>
</body>
</html>
```

- `input`을 두면 **없는 아이디를 입력해 `404` 에러 상태 + `[다시 시도]` 버튼**도 쉽게 테스트할 수 있습니다.

## 4. `script.js` — `state` + `render` + `fetchRepos` 흐름

```javascript
const usernameInput = document.querySelector('#username-input');
const loadBtn = document.querySelector('#load-btn');
const statusEl = document.querySelector('#status');
const gridEl = document.querySelector('#project-grid');

// 1. 상태(State)
const state = {
    status: 'idle', // 'idle' | 'loading' | 'success' | 'empty' | 'error'
    repos: [],
    errorMsg: '',
};

// 2. 화면 그리기(Render)
const render = () => {
    statusEl.innerHTML = '';
    gridEl.innerHTML = '';

    if (state.status === 'loading') {
        statusEl.textContent = '로딩 중...';
        return;
    }

    if (state.status === 'error') {
        statusEl.innerHTML = `
            <p>${state.errorMsg}</p>
            <button type="button" class="retry-btn">다시 시도</button>
        `;
        // 동적으로 만든 버튼에 addEventListener 연결 (onclick 금지 규칙 준수)
        document.querySelector('.retry-btn').addEventListener('click', () => {
            fetchRepos(usernameInput.value.trim());
        });
        return;
    }

    if (state.status === 'empty') {
        statusEl.textContent = '표시할 프로젝트가 없습니다.';
        return;
    }

    if (state.status === 'success') {
        gridEl.innerHTML = state.repos
            .map((repo) => {
                const { name, description, html_url, stargazers_count, language } = repo;
                return `
                    <article class="project-card">
                        <h3>${name}</h3>
                        <p>${description || '설명이 없습니다.'}</p>
                        <p>${language || '기타'} · ★ ${stargazers_count}</p>
                        <a href="${html_url}" target="_blank" rel="noopener noreferrer">GitHub 보기</a>
                    </article>
                `;
            })
            .join('');
    }
};

// 3. 비동기 데이터 요청
const fetchRepos = async (username) => {
    if (!username) return;

    state.status = 'loading';
    render();

    try {
        const response = await fetch(`https://api.github.com/users/${username}/repos`);
        if (!response.ok) {
            throw new Error(`불러오기 실패 (상태 코드: ${response.status})`);
        }

        const data = await response.json();
        state.repos = data;
        state.status = data.length === 0 ? 'empty' : 'success';
    } catch (error) {
        state.status = 'error';
        state.errorMsg = '프로젝트를 불러올 수 없습니다.';
        console.error(error);
    }

    render();
};

// 4. 이벤트 연결
loadBtn.addEventListener('click', () => {
    fetchRepos(usernameInput.value.trim());
});
```

## 5. 테스트해 볼 시나리오 4가지

1. **성공:** 본인 GitHub 아이디 입력 후 `[불러오기]` → 카드가 Grid로 잘 나오는지 확인
2. **에러:** 존재하지 않는 아이디 입력 후 `[불러오기]` → `"프로젝트를 불러올 수 없습니다."` + `[다시 시도]` 버튼이 뜨는지 확인
3. **빈 상태:** 저장소가 0개인 계정 또는 `data.filter(...)` 조건을 좁혀 `[]`이 되도록 테스트 → `"표시할 프로젝트가 없습니다."` 확인
4. **로딩:** Chrome F12 → **Network 탭 → Throttling을 `Slow 3G`**로 바꾸고 `[불러오기]` 클릭 → `"로딩 중..."`이 잠깐 보였다가 카드로 바뀌는지 확인
